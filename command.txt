git init – создание нового локального репрозитория
git add – добавление файла в репрозиторий
git status – проверка изменения  
git commit -m – добавить коммит к файлу или ветке
echo – создание нового файла с возможностью его заполнения	
git branch - создать новую ветку
git switch - перейти на ветку
git merge - слияние веток
git branch -d - удалить ветку
